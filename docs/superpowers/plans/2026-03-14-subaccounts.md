# Subaccounts Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Allow users to store multiple BingX API key pairs (subaccounts), switch between them, and tie each bot to a specific subaccount. Balance displayed per-subaccount.

**Architecture:** Rename `bingxApiKeys` to support multiple keys per user with a `label` field. Add `apiKeyId` FK to `tradingBots`. Create an account switcher component in the sidebar. All API calls and bot operations scope to the selected subaccount.

**Tech Stack:** Drizzle ORM migrations, Next.js API routes, React context for active account, HeroUI v3 components

**Dependency:** Requires Plan 1 (UI/UX Overhaul) to be completed first.

```typescript
export async function getBingxKeysByApiKeyId(apiKeyId: string): Promise<{ apiKey: string; secretKey: string } | null> {
  const row = await db.query.bingxApiKeys.findFirst({
    where: eq(bingxApiKeys.id, apiKeyId),
  });
  if (!row) return null;
  try {
    const secretKey = decryptSecret(row.secretKeyEncrypted);
    return { apiKey: row.apiKey, secretKey };
  } catch {
    return null;
  }
}

export async function getBingxClientByApiKeyId(apiKeyId: string): Promise<BingxClient | null> {
  const keys = await getBingxKeysByApiKeyId(apiKeyId);
  if (!keys) return null;
  return createBingxClient(keys.apiKey, keys.secretKey);
}
```

- [ ] **Step 2: Update saveBingxKeys to support labels**

```typescript
export async function saveBingxKeys(
  userId: string,
  apiKey: string,
  secretKey: string,
  label: string = 'Main',
  apiKeyId?: string
): Promise<string> {
  const trimmedApiKey = apiKey.trim();
  const trimmedSecretKey = secretKey.trim();
  const secretKeyEncrypted = encryptSecret(trimmedSecretKey);

  if (apiKeyId) {
    // Update existing key
    await db
      .update(bingxApiKeys)
      .set({
        apiKey: trimmedApiKey,
        secretKeyEncrypted,
        label,
        updatedAt: new Date(),
      })
      .where(and(eq(bingxApiKeys.id, apiKeyId), eq(bingxApiKeys.userId, userId)));
    return apiKeyId;
  }

  // Insert new key
  const [row] = await db
    .insert(bingxApiKeys)
    .values({
      userId,
      apiKey: trimmedApiKey,
      secretKeyEncrypted,
      label,
    })
    .returning({ id: bingxApiKeys.id });
  return row.id;
}
```

### Task 4: Create Active Account Context

**Files:**
- Create: `src/contexts/active-account.tsx`

- [ ] **Step 1: Create the context provider**

```tsx
'use client';

import { createContext, useContext, useState, useEffect, type ReactNode } from 'react';

type ApiKeyInfo = { id: string; label: string };

type ActiveAccountContextType = {
  accounts: ApiKeyInfo[];
  activeAccountId: string | null;
  setActiveAccountId: (id: string) => void;
  refreshAccounts: () => Promise<void>;
};

const ActiveAccountContext = createContext<ActiveAccountContextType | null>(null);

export function ActiveAccountProvider({ children }: { children: ReactNode }) {
  const [accounts, setAccounts] = useState<ApiKeyInfo[]>([]);
  const [activeAccountId, setActiveAccountId] = useState<string | null>(null);

  const refreshAccounts = async () => {
    try {
      const res = await fetch('/api/bingx/keys');
      const data = await res.json();
      const keys = Array.isArray(data) ? data : data.keys ?? [];
      setAccounts(keys);
      if (keys.length > 0 && !activeAccountId) {
        setActiveAccountId(keys[0].id);
      }
    } catch {}
  };

  useEffect(() => {
    refreshAccounts();
  }, []);

  return (
    <ActiveAccountContext.Provider
      value={{ accounts, activeAccountId, setActiveAccountId, refreshAccounts }}
    >
      {children}
    </ActiveAccountContext.Provider>
  );
}

export function useActiveAccount() {
  const ctx = useContext(ActiveAccountContext);
  if (!ctx) throw new Error('useActiveAccount must be used within ActiveAccountProvider');
  return ctx;
}
```

**Files:**

- [ ] **Step 1: Create the switcher component**

**IMPORTANT:** Check the actual HeroUI v3 Select API before implementing. The compound component syntax varies between HeroUI versions. The existing codebase uses `Select` + `Select.Trigger` + `Select.Value` + `Select.Indicator` + `Select.Popover` + `ListBox` + `ListBox.Item` pattern (see `src/components/trading/symbol-leverage-card.tsx` for the exact pattern). Match that pattern:

```tsx
'use client';

import { Select, ListBox } from '@heroui/react';
import { useActiveAccount } from '@/contexts/active-account';

export function AccountSwitcher() {
  const { accounts, activeAccountId, setActiveAccountId } = useActiveAccount();

  if (accounts.length === 0) return null;

  return (
    <div className="px-3 py-2">
      {/* Match the exact Select pattern used in symbol-leverage-card.tsx */}
      <Select
        aria-label="Select account"
        selectedKey={activeAccountId}
        onSelectionChange={(key) => {
          if (key !== null) setActiveAccountId(String(key));
        }}
      >
        <Select.Trigger className="w-full">
          <Select.Value />
          <Select.Indicator />
        </Select.Trigger>
        <Select.Popover>
          <ListBox>
            {accounts.map((acc) => (
              <ListBox.Item key={acc.id} id={acc.id}>
                {acc.label}
              </ListBox.Item>
            ))}
          </ListBox>
        </Select.Popover>
      </Select>
    </div>
  );
}
```
